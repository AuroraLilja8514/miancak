cmake_minimum_required(VERSION 3.22.1)
project(miancak LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake" ${CMAKE_MODULE_PATH})
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(SOURCES "${ROOT}/.native")
if(NOT EXISTS "${SOURCES}/librime/CMakeLists.txt")
  message(FATAL_ERROR "Run python3 scripts/prepare_rime.py before building")
endif()
add_compile_options("-ffile-prefix-map=${ROOT}=.")
add_link_options("LINKER:--build-id=none,-z,max-page-size=16384")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BOOST_INCLUDE_LIBRARIES algorithm crc dll interprocess preprocessor range regex scope_exit signals2 utility uuid vmd)
add_subdirectory("${SOURCES}/boost" boost EXCLUDE_FROM_ALL)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime/deps/yaml-cpp" yaml-cpp EXCLUDE_FROM_ALL)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime/deps/leveldb" leveldb EXCLUDE_FROM_ALL)
set(ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime/deps/marisa-trie" marisa EXCLUDE_FROM_ALL)
set(LIBMARISA marisa)
set(USE_SYSTEM_MARISA ON CACHE BOOL "" FORCE)
set(ENABLE_GTEST OFF CACHE BOOL "" FORCE)
set(ENABLE_BENCHMARK OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime/deps/opencc" opencc EXCLUDE_FROM_ALL)
if(NOT ANDROID)
  # Pinned OpenCC injects C++14 on GCC; pinned Marisa headers need C++17.
  target_compile_options(libopencc PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-std=c++17>)
endif()
file(GLOB OPENCC_HEADERS "${SOURCES}/librime/deps/opencc/src/*.hpp")
file(COPY ${OPENCC_HEADERS} DESTINATION "${CMAKE_BINARY_DIR}/include/opencc")
set(X11Keysym "${SOURCES}/librime/include")
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_LOGGING OFF CACHE BOOL "" FORCE)
set(ENABLE_EXTERNAL_PLUGINS OFF CACHE BOOL "" FORCE)
set(ENABLE_TIMESTAMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime" librime EXCLUDE_FROM_ALL)
# OpenCC's pinned public headers use std::iterator, deprecated in C++17.
# Keep upstream's other -Werror diagnostics; this is a toolchain compatibility warning.
target_compile_options(rime-static PRIVATE -Wno-error=deprecated-declarations)
set(RIME_SOURCE_DIR "${SOURCES}/librime")
set(rime_library rime-static)
set(rime_gears_library rime-static)
set(BUILD_TOOLS OFF CACHE BOOL "" FORCE)
include_directories("${SOURCES}/librime/src" "${SOURCES}/librime/include"
  "${CMAKE_CURRENT_BINARY_DIR}/librime/src")
add_subdirectory("${SOURCES}/octagram" octagram EXCLUDE_FROM_ALL)
add_subdirectory("${SOURCES}/miancak-lua-build" librime-lua EXCLUDE_FROM_ALL)
target_link_libraries(rime-octagram-objs PRIVATE rime-static)
target_link_libraries(rime-lua-objs PRIVATE rime-static)
target_compile_options(rime-lua-objs PRIVATE -Wno-error=deprecated-declarations)
target_include_directories(rime-lua-objs PRIVATE "${CMAKE_BINARY_DIR}/include")
add_library(miancak_rime SHARED rime_jni.cc)
target_include_directories(miancak_rime PRIVATE "${SOURCES}/librime/src" "${SOURCES}/octagram/src")
# Link the object files directly: module registration cannot be discarded from
# a static archive. External executable-plugin discovery remains disabled.
target_sources(miancak_rime PRIVATE $<TARGET_OBJECTS:rime-octagram-objs> $<TARGET_OBJECTS:rime-lua-objs>)
target_link_libraries(miancak_rime PRIVATE rime-static ${CMAKE_DL_LIBS})
if(NOT ANDROID)
  find_package(JNI REQUIRED)
  target_include_directories(miancak_rime PRIVATE ${JNI_INCLUDE_DIRS})
  add_executable(wanxiang_host_test tests/wanxiang_host_test.cc)
  target_include_directories(wanxiang_host_test PRIVATE "${SOURCES}/octagram/src")
  target_sources(wanxiang_host_test PRIVATE $<TARGET_OBJECTS:rime-octagram-objs> $<TARGET_OBJECTS:rime-lua-objs>)
  target_link_libraries(wanxiang_host_test PRIVATE rime-static ${CMAKE_DL_LIBS})
  add_executable(custom_config_test tests/custom_config_test.cc)
  target_link_libraries(custom_config_test PRIVATE yaml-cpp)
endif()
add_library(cdict_java SHARED
  "${ROOT}/vendor/cdict/libcdict/libcdict.c"
  "${ROOT}/vendor/cdict/java/jni/juloo_cdict_Cdict.c")
target_include_directories(cdict_java PRIVATE "${ROOT}/vendor/cdict/libcdict")
