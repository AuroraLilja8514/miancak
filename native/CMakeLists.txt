cmake_minimum_required(VERSION 3.22.1)
project(miancak LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(CMAKE_MODULE_PATH "${CMAKE_CURRENT_SOURCE_DIR}/cmake" ${CMAKE_MODULE_PATH})
get_filename_component(ROOT "${CMAKE_CURRENT_SOURCE_DIR}/.." ABSOLUTE)
set(SOURCES "${ROOT}/.native")
if(NOT EXISTS "${SOURCES}/librime/CMakeLists.txt")
  message(FATAL_ERROR "Run python3 scripts/prepare_rime.py before building")
endif()
add_compile_options("-ffile-prefix-map=${ROOT}=.")
add_link_options("LINKER:--build-id=none,-z,max-page-size=16384")
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
set(BOOST_INCLUDE_LIBRARIES algorithm crc dll interprocess preprocessor range regex scope_exit signals2 utility uuid vmd)
add_subdirectory("${SOURCES}/boost" boost EXCLUDE_FROM_ALL)
set(YAML_CPP_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(YAML_CPP_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime/deps/yaml-cpp" yaml-cpp EXCLUDE_FROM_ALL)
set(LEVELDB_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(LEVELDB_BUILD_BENCHMARKS OFF CACHE BOOL "" FORCE)
set(LEVELDB_INSTALL OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime/deps/leveldb" leveldb EXCLUDE_FROM_ALL)
set(ENABLE_TOOLS OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime/deps/marisa-trie" marisa EXCLUDE_FROM_ALL)
set(LIBMARISA marisa)
set(USE_SYSTEM_MARISA ON CACHE BOOL "" FORCE)
set(ENABLE_GTEST OFF CACHE BOOL "" FORCE)
set(ENABLE_BENCHMARK OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime/deps/opencc" opencc EXCLUDE_FROM_ALL)
file(GLOB OPENCC_HEADERS "${SOURCES}/librime/deps/opencc/src/*.hpp")
file(COPY ${OPENCC_HEADERS} DESTINATION "${CMAKE_BINARY_DIR}/include/opencc")
set(X11Keysym "${SOURCES}/librime/include")
set(BUILD_STATIC ON CACHE BOOL "" FORCE)
set(BUILD_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_LOGGING OFF CACHE BOOL "" FORCE)
set(ENABLE_EXTERNAL_PLUGINS OFF CACHE BOOL "" FORCE)
set(ENABLE_TIMESTAMP OFF CACHE BOOL "" FORCE)
add_subdirectory("${SOURCES}/librime" librime EXCLUDE_FROM_ALL)
add_library(miancak_rime SHARED rime_jni.cc)
target_include_directories(miancak_rime PRIVATE "${SOURCES}/librime/src")
target_link_libraries(miancak_rime PRIVATE rime-static)
add_library(cdict_java SHARED
  "${ROOT}/vendor/cdict/libcdict/libcdict.c"
  "${ROOT}/vendor/cdict/java/jni/juloo_cdict_Cdict.c")
target_include_directories(cdict_java PRIVATE "${ROOT}/vendor/cdict/libcdict")
